    return [...this.props.trace.iterations.values()].map((iter, i) => ({
        index: iter.iteration_index ?? i + 1,
        duration_ms: iter.duration_ms || 0,
        tokens: iter.tokens || { input: 0, output: 0, cache_read: 0, reasoning: 0, total: 0 },
    }));
}

- Add `traceTotals` getter that aggregates across all iterations (computed in LoopDetail since it has `props.trace` — do NOT call getTraceTotals from AiDebugApp per research anti-pattern):

```javascript
get traceTotals() {
    let total_input = 0, total_output = 0, total_cached = 0,
        total_reasoning = 0, total_duration_ms = 0;
    for (const iter of this.props.trace.iterations.values()) {
        const t = iter.tokens;
        if (t) {
            total_input += t.input || 0;
            total_output += t.output || 0;
            total_cached += t.cache_read || 0;
            total_reasoning += t.reasoning || 0;
        }
        total_duration_ms += iter.duration_ms || 0;
    }
    return { total_input, total_output, total_cached, total_reasoning, total_duration_ms };
}
```

Both getters read through the reactive proxy chain on `props.trace.iterations`, so OWL will re-render the table as new iterations arrive.

**1. In `loop_detail.js`:**
- Add OWL lifecycle imports: `import { Component, useRef, onMounted, onWillUnmount, onPatched } from "@odoo/owl";` (update the existing `import { Component } from "@odoo/owl"` line)
- Add `useRef` for the timer DOM element in `setup()`:

```javascript
setup() {
    // ... existing dialog setup ...
    this.timerRef = useRef("liveTimer");
    this._timerInterval = null;

    onMounted(() => {
        if (this.props.trace && this.props.trace.status === "running") {
            this._startTimer();
        }
    });

    onWillUnmount(() => {
        this._stopTimer();
    });

    // Watch for status transition: running → complete
    onPatched(() => {
        if (!this.props.trace) return;
        if (this.props.trace.status !== "running") {
            this._stopTimer();
        } else if (!this._timerInterval) {
            this._startTimer();
        }
    });
}
```

- Add `_startTimer()` and `_stopTimer()` methods:

```javascript
_startTimer() {
    // Immediately set initial value
    this._updateTimerDisplay();
    this._timerInterval = setInterval(() => this._updateTimerDisplay(), 1000);
}

_updateTimerDisplay() {
    if (!this.timerRef.el || !this.props.trace.started_at) return;
    const elapsed = Date.now() - this.props.trace.started_at.getTime();
    this.timerRef.el.textContent = formatDuration(elapsed);
}

_stopTimer() {
    if (this._timerInterval) {
        clearInterval(this._timerInterval);
        this._timerInterval = null;
    }
}
```

CRITICAL: Use `this.props.trace.started_at.getTime()` — `started_at` is a `Date` object (from `_placeTrace` and `hydrateTrace`), NOT a numeric timestamp. `Date.now() - started_at` would return NaN (Pitfall 3 from research).

CRITICAL: Use `setInterval` + `textContent` DOM mutation (not reactive state). Setting `useState({ elapsed })` and mutating it in `setInterval` would cause OWL to re-render the entire LoopDetail component (including Notebook and Metrics table) every second — this is the anti-pattern from research. DOM mutation via `useRef` is correct.

CRITICAL: Pair `_startTimer()` with `_stopTimer()` in `onWillUnmount` to prevent timer leaks when user navigates away (Pitfall 1 from research). The `onPatched` handler stops the timer when `status !== 'running'` (Pitfall 2 from research).
